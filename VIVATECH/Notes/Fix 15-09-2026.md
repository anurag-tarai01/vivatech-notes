@RequestMapping(value = "/amt01-transaction-report", method = RequestMethod.GET)  
@PreAuthorize("@permissionService.check('TRANSACTION_SINGLE_AMT01_REPORT')")  
public String getAmt01TransactionReport(  
        @RequestParam(name = "page", required = false) String page,  
        @RequestParam(name = "userId", required = false) String userId,  
        @RequestParam(name = "transactionStartDate", required = false) String transactionStartDate,  
        @RequestParam(name = "transactionEndDate", required = false) String transactionEndDate,  
        @RequestParam(name = "fromAccount", required = false) String fromAccount,  
        @RequestParam(name = "toAccount", required = false) String toAccount,  
        @RequestParam(name = "currency", required = false) String currency,  
        HttpServletRequest request, Model model) {  
  
    TransactionCalculationPaginationResponseDto responseDto = new TransactionCalculationPaginationResponseDto();  
    String selectedCurrency = StringUtils.isEmpty(currency) ? CURRENCY_CODE : currency;  
      
    if (transactionStartDate != null && !transactionStartDate.isEmpty() && transactionEndDate != null && !transactionEndDate.isEmpty()) {  
  
        String uri = apiGatewayAddress + "transaction-report/amt01-transaction-report";  
        int pageNumber = (page != null && !page.isEmpty()) ? Integer.parseInt(page) : 0;  
        Date formattedDateFrom = MfsUtils.stringToDate(transactionStartDate, "MMM dd yyyy hh:mm aa");  
        Date formattedDateTo = MfsUtils.stringToDate(transactionEndDate, "MMM dd yyyy hh:mm aa");  
  
        TransactionReportRequestDto requestDto = TransactionReportRequestDto.builder()  
                .pageNumber(pageNumber)  
                .transactionDateFrom(formattedDateFrom)  
                .transactionDateTo(formattedDateTo)  
                .fromAccount(fromAccount)  
                .toAccount(toAccount)  
                .userId(userId)  
                .transactionStatus("SUCCESS")  
                .transactionUserType(UserType.ADMIN)  
                .currency(selectedCurrency)  
                .build();  
        restTemplate.setInterceptors(SessionHelper.getInterceptor());  
        responseDto = restTemplate.postForObject(uri, requestDto, TransactionCalculationPaginationResponseDto.class);  
    } else {  
        responseDto.setContent(new ArrayList<>());  
    }  
    HashMap<String, String> parameterKeyValuePair = mfsUtils.getUrlParameterToKeyValuePair(request.getQueryString());  
    parameterKeyValuePair.put("currency", selectedCurrency);  
      
    List<SearchCriteria> searchCriteriaList = this.specificUserReportCriteria(parameterKeyValuePair);  
    searchCriteriaList.add(getFromAccountCriteria(parameterKeyValuePair));  
    searchCriteriaList.add(getToAccountCriteria(parameterKeyValuePair));  
    searchCriteriaList.add(mfsUtils.getCurrencyCriteria(parameterKeyValuePair, CURRENCY_CODE));  
  
    model.addAttribute("entries", responseDto);  
    model.addAttribute("searchCriteriaList", searchCriteriaList);  
    model.addAttribute("countryCode", Constants.COUNTRY_CODE);  
    model.addAttribute("title", VivaConstants.AMT01_TRANSACTION_REPORT);  
  
    return "admin/transaction-report/amt-transaction-report.html";  
}
